\section[RDD]{Regressão descontínua}
\subsection{Ideia e exemplos}

\begin{frame}{A ideia: um ``experimento natural'' no corte}
  \begin{itemize}
    \item Muitos programas têm uma \ind{regra de elegibilidade}: um índice contínuo e um \textbf{ponto de corte};
    \item Ex.: transferência de renda para quem tem renda per capita \textbf{até R\$ 77}. Uma família com R\$ 77 recebe; outra com R\$ 78, não;
    \item Perto do corte, as famílias são muito parecidas: é \destaque{como se houvesse um sorteio};
    \item O impacto é o tamanho do \ind{salto} no corte.
  \end{itemize}
\end{frame}

\begin{frame}{Regressão descontínua: características e exemplos}
  \carac{famílias \textbf{logo do outro lado do corte}}
        {\textbf{continuidade}: sem manipulação do corte e sem outros saltos no mesmo ponto}
        {efeito \textbf{local}, para quem está no corte}
        {índice contínuo, corte claro e \textbf{muitas observações} perto dele}
        {programas com elegibilidade por escore (renda, nota, idade, população)}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{México:} elegibilidade do Progresa e frequência escolar (Buddelmeyer e Skoufias, 2004).
      \item \textbf{Israel:} turmas divididas quando passam de 40 alunos (regra de Maimônides) e desempenho (Angrist e Lavy, 1999).
      \item \textbf{Eleições:} candidatos eleitos por margem mínima (Lee, 2008).
      \item \textbf{Hipotético:} transferência de renda com corte de R\$ 77 (script 05).
    \end{itemize}
\end{frame}

\begin{frame}{RDD, passo 1: como ler}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Antes de ler:} \texttt{x} $=$ renda $-$ 77, então $x=0$ é o corte; \texttt{elegivel} $=1$ para renda até R\$ 77; \texttt{janela} usa só famílias entre R\$ 62 e R\$ 92.
    \item \rot{\texttt{(Intercept)} $=68{,}4$:} frequência escolar de uma família \textbf{não elegível} bem no corte. \rot{\texttt{elegivel} $=6{,}2$:} quanto a família \textbf{elegível}, também no corte, frequenta a mais (74,6\%). Esse \textbf{salto é o efeito} (EP 0,9; $p<0{,}001$; verdade: 6).
    \item \rot{\texttt{x} e \texttt{elegivel:x}:} quanto a frequência muda a cada R\$ 1 de renda, de cada lado (a inclinação das retas). Não são efeito do programa.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}{RDD, passo 2: como ler}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Colunas:} \texttt{h} $=$ janela (h $=15$: famílias até R\$ 15 do corte); \texttt{efeito} $=$ o salto; \texttt{ep} $=$ erro-padrão; \texttt{n} $=$ famílias usadas; \texttt{li} e \texttt{ls} $=$ limites inferior e superior do IC 95\%.
    \item \rot{O dilema:} janela pequena ($h=5$) usa famílias muito parecidas, mas poucas (440): estimativa imprecisa (ep 1,56). Janela grande ($h=40$) usa muitas (2.833): mais precisa (ep 0,57), mas inclui famílias menos parecidas.
    \item \rot{Conclusão:} o efeito fica entre 5,5 e 7,1 e nenhum IC inclui zero: \textbf{não depende da janela}. Regra: mostrar várias janelas (Lee e Lemieux, 2010).
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}{RDD, passo 3: como ler}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Teste 1, idade da mãe:} o programa não muda a idade da mãe. Se ela ``saltasse'' no corte, as famílias dos dois lados seriam diferentes. Salto $=0{,}07$ ano, \texttt{Pr} $=0{,}93$ ($>0{,}05$): \textbf{sem salto}.
    \item \rot{Teste 2, contagem:} quantas famílias há em cada faixa de R\$ 5 de renda. Se as famílias ``baixassem'' a renda declarada para entrar, a faixa (72,77] ficaria cheia e a (77,82] vazia. Aqui: 213 $\times$ 227, normal (com manipulação simulada: 413 $\times$ 116).
    \item \rot{Regra:} covariáveis sem salto e contagens sem acúmulo no corte (McCrary, 2008).
  \end{itemize}
\end{leitura}
\end{frame}
