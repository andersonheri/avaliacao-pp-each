\begin{frame}{Regressão descontínua (RDD): um ``experimento natural''}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item Muitos programas têm uma \ind{regra de elegibilidade}: um índice e um \textbf{ponto de corte} (\emph{cutoff});
        \item Quem está \emph{logo abaixo} e \emph{logo acima} do corte é muito semelhante, mas só um grupo recebe a política;
        \item Perto do corte, é \destaque{como se houvesse um sorteio} --- que não está sob controle do avaliador;
        \item O impacto é o tamanho do \ind{salto} (descontinuidade) no corte.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Dois pré-requisitos}
        \small
        \begin{enumerate}
          \item Um índice de elegibilidade \textbf{contínuo};
          \item Um valor de corte \textbf{claramente definido}.
        \end{enumerate}
      \end{block}
      \vspace{0.1cm}
      \footnotesize Duas limitações: efeito \textbf{local} (só perto do corte) e necessidade de \textbf{muitas observações} perto do corte.
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), pp.~12--13.}
\end{frame}

\begin{frame}{Exemplo: transferência de renda com corte de pobreza}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Programa de transferência de renda: \textbf{elegível} quem tem renda per capita \textbf{até R\$ 77} (linha de extrema pobreza, exemplo do artigo);
        \item Uma família com R\$ 77 recebe; outra com R\$ 78 \emph{não}. Que diferença real existe entre elas?
        \item Dentro dessa faixa estreita, podemos tratar as famílias como \emph{distribuídas aleatoriamente} entre tratamento e controle.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Nosso mundo simulado}
        \small
        \begin{itemize}
          \item 4.000 famílias; resultado = \textbf{frequência escolar (\%)};
          \item Efeito verdadeiro: \destaque{+6 pontos percentuais};
          \item A frequência cresce com a renda: elegíveis são mais pobres e, sem programa, teriam frequência \emph{menor}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), pp.~12--13; Figura 7.}
\end{frame}

\begin{frame}{Testes de validade do desenho}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Três checagens}
        \small
        \begin{enumerate}
          \item \textbf{Covariáveis}: variáveis \emph{pré-tratamento} (ex.: idade da mãe) não devem ``pular'' no corte (aqui: salto de $0{,}07$, EP = 0,77);
          \item \textbf{Placebo}: cortes \emph{falsos} (R\$ 55, 65, 105) não devem gerar salto (aqui: $\approx 0$);
          \item \textbf{Manipulação}: as famílias conseguem ajustar a renda para entrar? Procure um \emph{acúmulo} logo abaixo do corte.
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\textwidth]{05_rdd_manipulacao.png}
    \end{column}
  \end{columns}
  \fonte{Simulação nossa (script 05).}
\end{frame}

\begin{frame}{Limites e variações do RDD}
  \begin{itemize}
    \item \textbf{Efeito local}: o resultado vale para famílias \emph{próximas do corte} (R\$ 77), não necessariamente para quem ganha R\$ 30;
    \item \textbf{Muitas observações} perto do corte são necessárias;
    \item \textbf{RDD ``fuzzy''}: quando nem todos os elegíveis recebem, usa-se (salto no resultado) $\div$ (salto na participação). Na simulação: $5{,}4\div0{,}78\approx 7{,}0$ (verdade: 6);
    \item \textbf{Não precisa ser ``regressão''}: o essencial é verificar se as observações acima e abaixo do corte estão \emph{balanceadas}.
  \end{itemize}
  \vspace{0.3cm}
  \begin{block}{Exemplo no artigo: Progresa (México)}
    \small Buddelmeyer e Skoufias (2004) comparam famílias elegíveis e inelegíveis no corte para avaliar o impacto na educação: aumento de \textbf{8,9\%} (1998) e \textbf{9,9\%} (1999) na presença escolar das \textbf{meninas}; para os meninos, +5\% sem significância.
  \end{block}
  \fonte{Batista e Domingos (2017), pp.~12--13.}
\end{frame}

d <- gerar_rdd()               # corte em R$ 77; x = renda - 77

# Regressao local: janela de +/- R$ 15, retas diferentes de cada lado
janela <- d[abs(d$x) <= 15, ]
m <- lm(freq ~ elegivel * x, data = janela)
coef(m)["elegivel"]                  # salto no corte: 6,2 p.p.

# Sensibilidade a janela
sapply(c(5, 15, 40), function(h) rdd_local(d, h)$efeito)

# Placebo: corte falso e teste de manipulacao (densidade da renda)
