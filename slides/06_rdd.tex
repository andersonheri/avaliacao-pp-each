\section[RDD]{Regressão descontínua}
\subsection{Ideia e exemplos}

\begin{frame}{A ideia: um ``experimento natural'' no corte}
  \begin{itemize}
    \item Muitos programas têm uma \ind{regra de elegibilidade}: um índice contínuo e um \textbf{ponto de corte};
    \item Ex.: transferência de renda para quem tem renda per capita \textbf{até R\$ 77}. Uma família com R\$ 77 recebe; outra com R\$ 78, não;
    \item Perto do corte, as famílias são muito parecidas: é \destaque{como se houvesse um sorteio};
    \item O impacto é o tamanho do \ind{salto} no corte.
  \end{itemize}
\end{frame}

\begin{frame}{Regressão descontínua: características e exemplos}
  \carac{famílias \textbf{logo do outro lado do corte}}
        {\textbf{continuidade}: sem manipulação do corte e sem outros saltos no mesmo ponto}
        {efeito \textbf{local}, para quem está no corte}
        {índice contínuo, corte claro e \textbf{muitas observações} perto dele}
        {programas com elegibilidade por escore (renda, nota, idade, população)}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{México:} elegibilidade do Progresa e frequência escolar (Buddelmeyer e Skoufias, 2004).
      \item \textbf{Israel:} turmas divididas quando passam de 40 alunos (regra de Maimônides) e desempenho (Angrist e Lavy, 1999).
      \item \textbf{Eleições:} candidatos eleitos por margem mínima (Lee, 2008).
      \item \textbf{Hipotético:} transferência de renda com corte de R\$ 77 (script 05).
    \end{itemize}
\end{frame}

\subsection{Passo a passo}

\begin{frame}[fragile]{RDD, passo 1: estimar o salto no corte}
\fazer{regressão local na janela de $\pm$ R\$ 15, com retas diferentes de cada lado (\texttt{x} = renda $-$ 77).}
\begin{consoleR}
@@> janela <- d[abs(d$x) <= 15, ]
@@> round(summary(lm(freq ~ elegivel * x, data = janela))$coefficients, 3)
            Estimate Std. Error t value Pr(>|t|)
(Intercept)   68.387      0.635 107.668    0.000
elegivel       6.179      0.895   6.902    0.000
x              0.054      0.072   0.750    0.454
elegivel:x     0.150      0.103   1.450    0.147
\end{consoleR}
\begin{leitura}
  \begin{itemize}
    \item \texttt{elegivel} $=6{,}18$ é o \textbf{salto no corte}: elegíveis no corte frequentam \destaque{6,2 p.p.} a mais (EP 0,90; $p<0{,}001$). Verdade: 6.
    \item \texttt{x} e \texttt{elegivel:x} são as inclinações de cada lado: \textbf{não} são efeitos.
    \item A janela é escolha do analista; em trabalho real, use a janela ótima de \texttt{rdrobust} (Calonico, Cattaneo e Titiunik, 2014).
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{RDD, passo 2: o resultado depende da janela?}
\fazer{reestimar com várias janelas (\texttt{h}) e comparar efeito, erro-padrão e tamanho da amostra.}
\begin{consoleR}
@@> round(sens, 2)
  h efeito   ep    n   li   ls
  5   6.59 1.56  440 3.54 9.65
 10   7.11 1.12  836 4.92 9.29
 15   6.18 0.90 1271 4.42 7.93
 20   5.91 0.78 1641 4.38 7.45
 30   5.50 0.65 2271 4.22 6.78
 40   5.64 0.57 2833 4.52 6.77
\end{consoleR}
\begin{leitura}
  \begin{itemize}
    \item O efeito fica entre 5,5 e 7,1; todos os IC (\texttt{li} a \texttt{ls}) excluem o zero: resultado \textbf{robusto}.
    \item Janela estreita: menos viés, mais incerteza (EP 1,56 com 440 famílias). Janela larga: o inverso (EP 0,57 com 2.833).
    \item \corte{relate várias janelas, incluindo metade e o dobro da escolhida (Lee e Lemieux, 2010).}
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{RDD, passo 3: o desenho é válido?}
\fazer{testar uma covariável pré-programa no corte e contar famílias perto do corte (manipulação).}
\begin{consoleRpeq}
@@> round(summary(lm(idade_mae ~ elegivel * x, data = janela))$coefficients["elegivel", ], 3)
  Estimate Std. Error    t value   Pr(>|t|)
     0.066      0.773      0.085      0.932
@@> table(cut(d$renda, breaks = c(62, 67, 72, 77, 82, 87, 92)))
(62,67] (67,72] (72,77] (77,82] (82,87] (87,92]
    197     206     213     227     190     238
\end{consoleRpeq}
\begin{leitura}
  \begin{itemize}
    \item A idade da mãe \textbf{não salta} no corte ($p=0{,}93$): as famílias dos dois lados são comparáveis.
    \item As contagens são parecidas dos dois lados de R\$ 77 (213 e 227): \textbf{sem acúmulo}. Com manipulação (simulada), seriam 413 e 116: RDD inválido.
    \item \corte{covariáveis sem salto ($p>0{,}05$) e densidade contínua no corte (McCrary, 2008; \texttt{rddensity}: Cattaneo, Jansson e Ma, 2020).}
  \end{itemize}
\end{leitura}
\end{frame}
