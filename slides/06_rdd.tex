\section[RDD]{Regressão descontínua}
\subsection{Ideia e exemplos}

\begin{frame}{A ideia: um ``experimento natural'' no corte}
  \begin{itemize}
    \item Muitos programas têm uma \ind{regra de elegibilidade}: um índice contínuo e um \textbf{ponto de corte};
    \item Ex.: transferência de renda para quem tem renda per capita \textbf{até R\$ 77}. Uma família com R\$ 77 recebe; outra com R\$ 78, não;
    \item Perto do corte, as famílias são muito parecidas: é \destaque{como se houvesse um sorteio};
    \item O impacto é o tamanho do \ind{salto} no corte.
  \end{itemize}
\end{frame}

\begin{frame}{Regressão descontínua: características e exemplos}
  \carac{famílias \textbf{logo do outro lado do corte}}
        {\textbf{continuidade}: sem manipulação do corte e sem outros saltos no mesmo ponto}
        {efeito \textbf{local}, para quem está no corte}
        {índice contínuo, corte claro e \textbf{muitas observações} perto dele}
        {programas com elegibilidade por escore (renda, nota, idade, população)}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{México:} elegibilidade do Progresa e frequência escolar (Buddelmeyer e Skoufias, 2004).
      \item \textbf{Israel:} turmas divididas quando passam de 40 alunos (regra de Maimônides) e desempenho (Angrist e Lavy, 1999).
      \item \textbf{Eleições:} candidatos eleitos por margem mínima (Lee, 2008).
    \end{itemize}
\end{frame}

\subsection{Passo a passo}

\begin{frame}[fragile]{RDD, passo 1: de quanto é o salto no corte?}
\begin{consoleR}
@@> janela <- d[abs(d$x) <= 15, ]
@@> m <- lm(freq ~ elegivel * x, data = janela)
@@> round(summary(m)$coefficients["elegivel", ], 2)
  Estimate Std. Error    t value   Pr(>|t|)
      6.18       0.90       6.90       0.00
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} \texttt{x} é a renda menos 77 (o corte é $x=0$); \texttt{janela} usa famílias entre R\$ 62 e R\$ 92. \texttt{Estimate} é o salto: \destaque{6,2 pontos} de frequência escolar (verdade: 6), EP 0,9.
    \item \rot{Corte:} $p<0{,}05$ ($t$ grande, $p\approx 0$).
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{RDD, passo 2: o resultado depende da janela?}
\begin{consoleRpeq}
@@> round(sens[sens$h %in% c(5, 15, 40), ], 2)
   h efeito   ep    n   li   ls
2  5   6.59 1.56  440 3.54 9.65
5 15   6.18 0.90 1271 4.42 7.93
8 40   5.64 0.57 2833 4.52 6.77
@@> sapply(c(h5 = 5, h15 = 15, h40 = 40), function(h)
@@+        table(d$elegivel[abs(d$x) <= h]))
   h5 h15  h40
0 227 655 1684
1 213 616 1149
\end{consoleRpeq}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} \texttt{h} é a janela em R\$; \texttt{n}, as famílias; a tabela de baixo separa controle (\texttt{0}) e tratamento (\texttt{1}). Janela pequena: poucas famílias (ep 1,56); grande: mais precisão (ep 0,57).
    \item \rot{Corte:} várias janelas (Lee e Lemieux, 2010); IC sem zero.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{RDD, passo 3: os dois lados do corte são comparáveis?}
\begin{consoleR}
@@> m2 <- lm(idade_mae ~ elegivel * x, data = janela)
@@> round(summary(m2)$coefficients["elegivel", ], 2)
  Estimate Std. Error    t value   Pr(>|t|)
      0.07       0.77       0.09       0.93
@@> table(cut(d$renda, breaks = c(62, 67, 72, 77, 82, 87, 92)))
(62,67] (67,72] (72,77] (77,82] (82,87] (87,92]
    197     206     213     227     190     238
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} (1) a idade da mãe não deve ``saltar'' no corte: salto $0{,}07$, $p=0{,}93$. (2) a contagem por faixa não deve acumular logo abaixo do corte: 213 e 227, normal.
    \item \rot{Corte:} covariáveis sem salto e densidade contínua no corte (McCrary, 2008).
  \end{itemize}
\end{leitura}
\end{frame}
