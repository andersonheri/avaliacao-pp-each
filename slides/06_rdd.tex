\section[RDD]{Regressão descontínua}

\begin{frame}{A ideia: um ``experimento natural'' no corte}
  \begin{itemize}
    \item Muitos programas têm uma \ind{regra de elegibilidade}: um índice contínuo e um \textbf{ponto de corte};
    \item Ex.: transferência de renda para quem tem renda per capita \textbf{até R\$ 77}. Uma família com R\$ 77 recebe; outra com R\$ 78, não;
    \item Perto do corte, as famílias são muito parecidas: é \destaque{como se houvesse um sorteio};
    \item O impacto é o tamanho do \ind{salto} no corte.
  \end{itemize}
  \vspace{0.2cm}
  \small Cuidados: o efeito é \textbf{local} (só perto do corte) e exige \textbf{muitas observações} ali.
\end{frame}

\begin{frame}{Testes de validade}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \footnotesize
      \begin{enumerate}
        \item \textbf{Covariáveis} pré-tratamento não devem ``pular'' no corte;
        \item \textbf{Placebo}: cortes falsos não devem gerar salto;
        \item \textbf{Manipulação}: as famílias conseguem ajustar a renda para entrar? Procure acúmulo logo abaixo do corte.
      \end{enumerate}
    \end{column}
    \begin{column}{0.52\textwidth}
      \includegraphics[width=\textwidth]{05_rdd_manipulacao.png}
    \end{column}
  \end{columns}
  \vspace{0.2cm}
  \scriptsize Exemplo real: Progresa, no México (Buddelmeyer e Skoufias, 2004).
\end{frame}

\begin{lstlisting}
d <- gerar_rdd()           # corte em R$ 77; x = renda - 77

# Regressao local: janela de +/- R$ 15, retas diferentes por lado
janela <- d[abs(d$x) <= 15, ]
m <- lm(freq ~ elegivel * x, data = janela)
coef(m)["elegivel"]                     # salto no corte: 6,2 p.p.

# Sensibilidade a janela
sapply(c(5, 15, 40), function(h) rdd_local(d, h)$efeito)

# Teste de manipulacao: histograma da renda em torno do corte
hist(d$renda, breaks = 60); abline(v = 77)
\end{lstlisting}
