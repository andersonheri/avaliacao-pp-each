\begin{frame}{O que vamos fazer}
  \begin{itemize}
    \item Entender por que \destaque{comparar} participantes e não participantes (ou antes e depois) \destaque{engana};
    \item Conhecer as principais técnicas \textbf{quantitativas} de avaliação de impacto, com exemplos de políticas públicas e um \ind{passo a passo} em R;
    \item Ver o que a avaliação \ind{qualitativa} e os \ind{métodos mistos} acrescentam aos números;
    \item Metade do tempo para cada lado: \textbf{quanto} (até 15:40) e \textbf{como e por quê} (depois do intervalo).
  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{Como funciona}
    \small Aula expositiva + scripts em R comentados (demonstração). Os dados são simulados: sabemos a \emph{verdade} e vemos se a técnica a recupera.
  \end{block}
\end{frame}

\begin{frame}{Para começar: o programa funciona?}
  \begin{exerciciobox}{Enquete rápida (chat)}
    \small
    Um programa de \textbf{qualificação profissional} foi oferecido a trabalhadores que se inscreveram voluntariamente. Um ano depois, quem participou ganha, em média, \destaque{R\$ 587 a mais} por mês do que quem não participou.
  \end{exerciciobox}
  \vspace{0.2cm}
  O programa aumentou a renda dos participantes em R\$ 587?
  \begin{itemize}
    \item[(a)] Sim, o efeito é de R\$ 587
    \item[(b)] O efeito é menor que isso
    \item[(c)] Não dá para saber só com essa informação
  \end{itemize}
\end{frame}

\subsection{Tipos de avaliação}

\begin{frame}{Tipos de avaliação: pela pergunta}
  \footnotesize
  \renewcommand{\arraystretch}{1.2}
  \begin{tabularx}{\textwidth}{@{}p{1.9cm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}
    \toprule
    & \textbf{Pergunta} & \textbf{Exemplo: curso de qualificação} \\ \midrule
    \ind{Diagnóstico} & Qual é o problema e quem é afetado? & Quantos trabalhadores estão desempregados e por quê? \\
    \ind{Processo} & As atividades aconteceram e chegaram ao público-alvo? & Quantos se inscreveram? Os cursos ocorreram? \\
    \ind{Resultados} & A situação do público-alvo mudou? & A renda dos participantes subiu entre antes e depois? \\
    \rowcolor{laranjaclaro}
    \destaque{Impacto} & A mudança \textbf{é causada} pelo programa? & Quanto da renda existe \emph{por causa} do curso? \\
    \ind{Eficiência} & Os resultados justificam o custo? & Quanto custa cada trabalhador a mais empregado? \\ \bottomrule
  \end{tabularx}
\end{frame}

\begin{frame}{Tipos de avaliação: por função e por quem avalia}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Quanto à função}
        \footnotesize
        \begin{itemize}
          \item \textbf{Formativa}: melhorar a política \emph{enquanto} ela acontece;
          \item \textbf{Somativa}: julgar o resultado final e decidir sobre sua continuidade.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{block}{Quanto a quem avalia}
        \footnotesize
        \begin{itemize}
          \item \textbf{Interna}: conhece a política, mas pode ter conflito de interesse;
          \item \textbf{Externa}: mais independente, conhece menos o contexto;
          \item \textbf{Participativa}: beneficiários e gestores ajudam a desenhar e interpretar.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  \centering\small Nenhuma é a ``melhor'': depende do \textbf{uso} que se pretende dar à avaliação.
\end{frame}

\begin{frame}{Monitoramento e avaliação não são a mesma coisa}
  \footnotesize
  \renewcommand{\arraystretch}{1.25}
  \begin{tabularx}{\textwidth}{@{}p{2.2cm}>{\raggedright\arraybackslash}X>{\raggedright\arraybackslash}X@{}}
    \toprule
    & \textbf{Monitoramento} & \textbf{Avaliação} \\ \midrule
    Pergunta & Estamos fazendo o que planejamos? & Funcionou? Por quê? Para quem? \\
    Frequência & Contínua (mensal, trimestral) & Pontual, em momentos-chave \\
    Dados & Indicadores e registros administrativos & Pesquisas, experimentos, entrevistas \\
    Causalidade & Não estabelece & Pode estabelecer (impacto) \\ \bottomrule
  \end{tabularx}
  \vspace{0.3cm}\par
  \small Um bom monitoramento \emph{alimenta} a avaliação: sem registros de quem participou e quando, não há como estimar impacto depois.
\end{frame}
