% ---- Objetivos ----------------------------------------------------------------------
\begin{frame}{O que você vai levar daqui}
  \begin{enumerate}
    \item \ind{Explicar} por que comparar participantes e não participantes (ou antes e depois) \destaque{engana};
    \item \ind{Reconhecer} o que cada técnica quantitativa assume para identificar um efeito causal;
    \item \ind{Simular e estimar} em R o impacto por experimento, pareamento, DiD e RDD;
    \item \ind{Diagnosticar} quando uma estimativa não é confiável (balanceamento, tendências, manipulação);
    \item \ind{Combinar} números e narrativas: o que as técnicas qualitativas acrescentam.
  \end{enumerate}
  \vspace{0.4cm}
  \begin{block}{Ideia-guia do artigo}
    \small Técnicas quantitativas dizem \emph{quanto} a política mudou o resultado; técnicas qualitativas explicam \emph{como} e \emph{por quê}. Avaliações completas combinam as duas.
  \end{block}
\end{frame}

% ---- Evidência e decisão ------------------------------------------------------------
\begin{frame}{Por que avaliar? Políticas baseadas em evidências}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \item Recursos e capacidade estatal são \alert{limitados}: é preciso \ind{escolher} que políticas manter, reformular ou interromper;
        \item Avaliar produz o critério: \emph{o problema foi resolvido? O programa funcionou?}
        \item Esse raciocínio é o das \ind{políticas baseadas em evidências} (\emph{evidence-based policy-making});
        \item A avaliação é o estágio de \destaque{renovação} do ciclo de políticas.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{beamercolorbox}[wd=\textwidth,rounded=true,sep=0.35cm]{laranjabox}
        \itshape\small
        Médicos usam evidências quando prescrevem tratamentos. Formuladores de políticas deveriam usar também.
        \par\raggedleft\upshape\scriptsize --- \emph{The Economist}, em epígrafe do artigo
      \end{beamercolorbox}
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), pp.~1--2.}
\end{frame}

% ---- Tipos de avaliação ---------------------------------------------------------------
\begin{frame}{Há várias formas de classificar uma avaliação}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{Quanto ao \textit{tempo}}
        \small
        \begin{itemize}
          \item \textbf{Ex-ante}: antes da implementação (desenho, viabilidade)
          \item \textbf{Ex-post}: depois (efeitos observados)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Quanto a \textit{quem avalia}}
        \small
        \begin{itemize}
          \item \textbf{Interna}: gestores do programa
          \item \textbf{Externa}: avaliadores independentes
          \item \textbf{Participativa}: beneficiários contribuem com seu \emph{input}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Quanto ao \textit{objeto}}
        \small
        \begin{itemize}
          \item \textbf{Processos}
          \item \textbf{Resultados}
          \item \textbf{Impactos}
          \item \textbf{Eficiência}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4cm}
  \begin{center}
    \destaque{Nosso foco hoje: avaliação de impacto ex-post} (e como pensá-la desde o desenho da política).
  \end{center}
  \fonte{Batista e Domingos (2017), p.~2, citando Cohen e Franco (2013) e Rossi, Lipsey e Freeman (2004).}
\end{frame}

% ---- Processo, resultado, impacto ----------------------------------------------------------
\begin{frame}{Processo, resultados e impacto}
  \footnotesize
  \renewcommand{\arraystretch}{1.25}
  \begin{tabularx}{\textwidth}{@{}p{2.1cm}XX@{}}
    \toprule
    & \textbf{Pergunta} & \textbf{No programa Qualifica} \\ \midrule
    \ind{Processo} & As atividades foram realizadas e os serviços chegaram ao público-alvo? (cobertura, vieses na entrega) & Quantos foram inscritos? Os cursos aconteceram? \\
    \ind{Resultados} (\emph{outcome}) & A situação do público-alvo mudou? (metas pré-estabelecidas) & A renda dos participantes aumentou entre antes e depois? \\
    \rowcolor{laranjaclaro}
    \destaque{Impacto} & A mudança \textbf{é causada} pelo programa? Qual seria o resultado \emph{sem} ele? & Quanto da variação de renda existe \emph{por causa} do treinamento? \\ \bottomrule
  \end{tabularx}
  \vspace{0.15cm}
  \begin{block}{Atenção}
    \footnotesize Resultado (``a renda subiu'') \textbf{não é} impacto (``a renda subiu por causa do programa''): é preciso isolar o efeito do programa de outros fatores.
  \end{block}
  \fonte{Batista e Domingos (2017), p.~3; Rossi, Lipsey e Freeman (2004).}
\end{frame}

\begin{frame}{Por que a cadeia importa para a avaliação de impacto}
  \begin{itemize}
    \item Formular bem a pergunta exige dizer \ind{quais resultados} o programa busca e \ind{por quais canais} (a ``teoria da mudança'');
    \item O que está sob o controle da agência (insumos $\to$ produtos) é \textbf{implementação};
    \item Os resultados dependem da interação entre o produto e o \destaque{contexto e as particularidades dos beneficiários} --- fora do controle direto do gestor;
    \item A avaliação de impacto \alert{isola o efeito do programa} de outros fatores: flutuações econômicas, mudanças nas características dos beneficiários.
  \end{itemize}
  \vspace{0.3cm}
  \begin{block}{A pergunta causal}
    \small Afirmar que o resultado final observado ``deve-se ao programa'' é afirmar que, \emph{sem o programa, o resultado não teria sido observado}. Precisamos, portanto, do \destaque{contrafactual}.
  \end{block}
  \fonte{Batista e Domingos (2017), pp.~3--4.}
\end{frame}
