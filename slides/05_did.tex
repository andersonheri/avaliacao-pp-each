\section[DiD]{Diferença-em-diferenças}

\begin{frame}{A ideia: comparar variações}
  \begin{block}{Dupla diferença}
    \small
    $$\text{DD}=(\bar Y_{T}^{\,depois}-\bar Y_{T}^{\,antes})-(\bar Y_{C}^{\,depois}-\bar Y_{C}^{\,antes})$$
  \end{block}
  \begin{itemize}
    \item A variação do \textbf{controle} é a tendência que o tratado teria seguido sem a política;
    \item Elimina diferenças \emph{fixas} entre os grupos, mesmo as não observadas;
    \item Pressuposto central: \destaque{tendências paralelas}.
  \end{itemize}
\end{frame}

\begin{frame}{Exemplo numérico}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      \footnotesize
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{lrr}
        \toprule
        \textbf{Renda} & \textbf{Part.} & \textbf{Não} \\ \midrule
        Antes & 80 & 90 \\
        Depois & 125 & 120 \\ \midrule
        Variação & +45 & +30 \\ \bottomrule
      \end{tabular}
      \vspace{0.3cm}

      DD $=45-30=$ \destaque{15}\\[4pt]
      {\scriptsize Antes-depois: 45 (superestima). Participantes $\times$ não: 5 (subestima).}
    \end{column}
    \begin{column}{0.6\textwidth}
      \includegraphics[width=\textwidth]{04_figura6.png}
    \end{column}
  \end{columns}
  \fonte{Exemplo adaptado de Khandker, Koolwal e Samad (2010).}
\end{frame}

\begin{frame}{Um caso real: Lei Maria da Penha}
  \begin{itemize}
    \item Benefícios para \emph{todas} as mulheres do país: como achar um controle?
    \item Cerqueira et al.\ (2015): \textbf{homens} como controle, expostos às outras causas gerais de violência, mas \emph{não beneficiários};
    \item Resultado: taxa de homicídios; DD = interação entre ser mulher e o período pós-lei. A lei mostrou-se \destaque{efetiva}.
  \end{itemize}
  \vspace{0.2cm}
  \begin{exerciciobox}{Para pensar}
    \small Que evento poderia afetar mulheres e homens de forma diferente e violar as tendências paralelas?
  \end{exerciciobox}
\end{frame}

\begin{frame}[fragile]{Hora do R: \texttt{04\_diferenca\_em\_diferenca.R}}
\begin{lstlisting}
painel <- gerar_painel()      # 1.000 trabalhadores x 6 trimestres

# DD "na mao": quatro medias
m <- function(t, p) mean(painel$renda[painel$tratado == t &
                                      painel$pos == p])
(m(1, 1) - m(1, 0)) - (m(0, 1) - m(0, 0))      # 300

# DD por regressao: coeficiente da interacao
summary(lm(renda ~ tratado * pos, data = painel))

# Efeitos fixos e erro-padrao por cluster
library(fixest)
feols(renda ~ tratado:pos | id + trimestre,
      data = painel, cluster = ~id)
\end{lstlisting}
\end{frame}
