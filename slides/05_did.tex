\section[DiD]{Diferença-em-diferenças}
\subsection{Ideia e exemplos}

\begin{frame}{A ideia: comparar variações}
  \begin{block}{Dupla diferença}
    \small
    $$\text{DD}=(\bar Y_{T}^{\,depois}-\bar Y_{T}^{\,antes})-(\bar Y_{C}^{\,depois}-\bar Y_{C}^{\,antes})$$
  \end{block}
  \begin{itemize}
    \item A variação do \textbf{controle} é a tendência que o tratado teria seguido sem a política;
    \item Elimina diferenças \emph{fixas} entre os grupos, mesmo as não observadas;
    \item Pressuposto central: \destaque{tendências paralelas}.
  \end{itemize}
\end{frame}

\begin{frame}{Exemplo numérico}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      \footnotesize
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{lrr}
        \toprule
        \textbf{Renda} & \textbf{Part.} & \textbf{Não} \\ \midrule
        Antes & 80 & 90 \\
        Depois & 125 & 120 \\ \midrule
        Variação & +45 & +30 \\ \bottomrule
      \end{tabular}
      \vspace{0.3cm}

      DD $=45-30=$ \destaque{15}\\[4pt]
      {\scriptsize Antes-depois: 45 (superestima). Participantes $\times$ não: 5 (subestima).}
    \end{column}
    \begin{column}{0.6\textwidth}
      \includegraphics[width=\textwidth]{04_figura6.png}
    \end{column}
  \end{columns}
  \fonte{Exemplo adaptado de Khandker, Koolwal e Samad (2010).}
\end{frame}

\begin{frame}{Diferença-em-diferenças: características e exemplos}
  \carac{a \textbf{tendência do grupo de controle} aplicada ao tratado}
        {\textbf{tendências paralelas} na ausência da política; sem antecipação; sem contaminação}
        {efeito sobre os tratados (ATT)}
        {dados \textbf{antes e depois} de tratados e controles (painel ou cortes repetidos)}
        {política implantada só em parte do território ou da população; leis}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{Brasil:} Lei Maria da Penha. Mulheres (tratamento) $\times$ homens (controle), taxa de homicídios (Cerqueira et al., 2015).
      \item \textbf{EUA:} aumento do salário mínimo em Nova Jersey, com lanchonetes da Pensilvânia como controle (Card e Krueger, 1994).
      \item \textbf{Hipotético:} Qualifica implantado em uma região (script 04).
    \end{itemize}
\end{frame}

\subsection{Passo a passo}

\begin{frame}[fragile]{DiD, passo 1: quanto cada grupo mudou?}
\begin{consoleR}
@@> aggregate(renda ~ tratado + pos, data = painel, FUN = mean)
  tratado pos    renda
1       0   0 2170.783
2       1   0 1868.194
3       0   1 2405.008
4       1   1 2402.652
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Como ler as colunas:} \texttt{tratado} 0 = controle, 1 = região do programa; \texttt{pos} 0 = antes, 1 = depois; \texttt{renda} = média do grupo naquele período.
    \item \rot{Controle} (linhas 1 e 3): $2.171\to2.405$, subiu \textbf{234}. É o que aconteceria sem o programa (a tendência).
    \item \rot{Tratados} (linhas 2 e 4): $1.868\to2.403$, subiu \textbf{534}.
    \item \rot{Efeito:} o que os tratados cresceram \emph{além} do normal: $534-234=$ \destaque{300}. Comparar só o ``depois'' (2.403 $\times$ 2.405) daria zero, porque os tratados eram mais pobres.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{DiD, passo 2: a mesma conta em forma de regressão}
\begin{consoleR}
@@> round(summary(lm(renda ~ tratado * pos, data = painel))$coefficients, 2)
            Estimate Std. Error t value Pr(>|t|)
(Intercept)  2170.78       8.91  243.54        0
tratado      -302.59      14.09  -21.47        0
pos           234.22      12.61   18.58        0
tratado:pos   300.23      19.93   15.06        0
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Cada linha repete uma conta do passo 1:} \texttt{(Intercept)} $=2.171$: controle antes. \texttt{tratado} $=-302$: os tratados eram R\$ 302 mais pobres antes. \texttt{pos} $=+234$: quanto o controle cresceu. \texttt{tratado:pos} $=+300$: o crescimento \emph{extra} dos tratados, \textbf{o efeito}.
    \item \rot{Por que usar regressão:} ela dá o erro-padrão e permite incluir controles.
    \item \rot{Mas cuidado:} este erro-padrão (19,9) trata os 6 trimestres de uma mesma pessoa como se fossem pessoas diferentes. Isso está errado; o passo 3 corrige.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{DiD, passo 3: qual a incerteza correta?}
\begin{consoleR}
@@> feols(renda ~ tratado:pos | id + trimestre, data = painel, cluster = ~id)
OLS estimation, Dep. Var.: renda
Observations: 6,000
Fixed-effects: id: 1,000,  trimestre: 6
Standard-errors: Clustered (id)
            Estimate Std. Error t value  Pr(>|t|)
tratado:pos  300.234    7.76534 38.6633 < 2.2e-16 ***
---
Signif. codes:  0 '***' 0.001 '**' 0.01 '*' 0.05 '.' 0.1 ' ' 1
RMSE: 135.3     Adj. R2: 0.881241
              Within R2: 0.227944
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{O que mudou no comando:} \texttt{| id + trimestre} compara cada trabalhador com ele mesmo ao longo do tempo; \texttt{cluster = \textasciitilde id} avisa que as observações da mesma pessoa se parecem.
    \item \rot{Onde olhar:} a linha \texttt{tratado:pos}. Efeito $=300{,}2$; erro-padrão $=7{,}8$; IC 95\% $\approx 300\pm2\times7{,}8$, de 285 a 315; \texttt{***}: $p<0{,}001$. RMSE e R2 medem o ajuste, não o efeito.
    \item \rot{Regra:} agrupe o erro-padrão pela unidade que se repete no tempo (Bertrand, Duflo e Mullainathan, 2004).
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{DiD, passo 4: os grupos já se afastavam antes?}
\begin{consoleR}
@@> estudo <- feols(renda ~ i(trimestre, tratado, ref = 3) | id + trimestre,
@@+                 data = painel, cluster = ~id)
@@> round(coeftable(estudo), 2)
                     Estimate Std. Error t value Pr(>|t|)
trimestre::1:tratado    23.15      14.27    1.62     0.11
trimestre::2:tratado    20.67      13.59    1.52     0.13
trimestre::4:tratado   317.24      13.66   23.23     0.00
trimestre::5:tratado   306.06      13.61   22.48     0.00
trimestre::6:tratado   321.22      13.36   24.04     0.00
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Por que testar:} se os grupos já se afastavam antes, o DiD culparia o programa por uma diferença que já existia.
    \item \rot{Onde olhar:} cada linha é a distância entre tratados e controle naquele trimestre, comparada ao 3.º (o último antes). Trimestres 1 e 2 (antes): 23 e 21, com \texttt{Pr} 0,11 e 0,13 ($>0{,}05$): \textbf{não diferem de zero}. Trimestres 4 a 6 (depois): ${\approx}$ 300.
    \item \rot{Regra:} coeficientes de antes com $p>0{,}05$ e pequenos perto do efeito (23 $\times$ 300). ``Não significativo'' não prova o paralelismo (Roth, 2022).
  \end{itemize}
\end{leitura}
\end{frame}
