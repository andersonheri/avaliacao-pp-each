\section[DiD]{Diferença-em-diferenças}
\subsection{Ideia e exemplos}

\begin{frame}{A ideia: comparar variações}
  \begin{block}{Dupla diferença}
    \small
    $$\text{DD}=(\bar Y_{T}^{\,depois}-\bar Y_{T}^{\,antes})-(\bar Y_{C}^{\,depois}-\bar Y_{C}^{\,antes})$$
  \end{block}
  \begin{itemize}
    \item A variação do \textbf{controle} é a tendência que o tratado teria seguido sem a política;
    \item Elimina diferenças \emph{fixas} entre os grupos, mesmo as não observadas;
    \item Pressuposto central: \destaque{tendências paralelas}.
  \end{itemize}
\end{frame}

\begin{frame}{Exemplo numérico}
  \begin{columns}[T]
    \begin{column}{0.4\textwidth}
      \footnotesize
      \renewcommand{\arraystretch}{1.3}
      \begin{tabular}{lrr}
        \toprule
        \textbf{Renda} & \textbf{Part.} & \textbf{Não} \\ \midrule
        Antes & 80 & 90 \\
        Depois & 125 & 120 \\ \midrule
        Variação & +45 & +30 \\ \bottomrule
      \end{tabular}
      \vspace{0.3cm}

      DD $=45-30=$ \destaque{15}\\[4pt]
      {\scriptsize Antes-depois: 45 (superestima). Participantes $\times$ não: 5 (subestima).}
    \end{column}
    \begin{column}{0.6\textwidth}
      \includegraphics[width=\textwidth]{04_figura6.png}
    \end{column}
  \end{columns}
  \fonte{Exemplo adaptado de Khandker, Koolwal e Samad (2010).}
\end{frame}

\begin{frame}{Diferença-em-diferenças: características e exemplos}
  \carac{a \textbf{tendência do grupo de controle} aplicada ao tratado}
        {\textbf{tendências paralelas} na ausência da política; sem antecipação; sem contaminação}
        {efeito sobre os tratados (ATT)}
        {dados \textbf{antes e depois} de tratados e controles (painel ou cortes repetidos)}
        {política implantada só em parte do território ou da população; leis}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{Brasil:} Lei Maria da Penha. Mulheres (tratamento) $\times$ homens (controle), taxa de homicídios (Cerqueira et al., 2015).
      \item \textbf{EUA:} aumento do salário mínimo em Nova Jersey, com lanchonetes da Pensilvânia como controle (Card e Krueger, 1994).
      \item \textbf{Brasil:} expansão da Estratégia Saúde da Família entre municípios e mortalidade, sobretudo nas primeiras idades (Rocha e Soares, 2010).
    \end{itemize}
\end{frame}

\subsection{Passo a passo}

\begin{frame}[fragile]{DiD, passo 1: quanto cada grupo mudou?}
\begin{consoleR}
@@> aggregate(renda ~ tratado + pos, data = painel, FUN = mean)
  tratado pos    renda
1       0   0 2170.783
2       1   0 1868.194
3       0   1 2405.008
4       1   1 2402.652
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} \texttt{tratado} 1 = região do programa; \texttt{pos} 1 = depois. Controle: $2.171\to2.405$ (subiu \textbf{234}). Tratados: $1.868\to2.403$ (subiu \textbf{534}).
    \item \rot{Efeito:} $534-234=$ \destaque{300}. Comparar só o ``depois'' daria zero, porque os tratados eram mais pobres.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{DiD, passo 2: qual a incerteza correta?}
\begin{consoleR}
@@> round(summary(reg_dd)$coefficients["tratado:pos", ], 2)
  Estimate Std. Error    t value   Pr(>|t|)
    300.23      19.93      15.06       0.00
@@> round(coeftable(reg_fe)[, 1:2], 2)
            Estimate Std. Error
tratado:pos   300.23       7.77
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} o efeito é o mesmo (\destaque{300}), mas o erro-padrão muda. O primeiro (19,9) trata os 6 trimestres da mesma pessoa como pessoas diferentes; o segundo (7,8) compara cada trabalhador com ele mesmo e agrupa por pessoa.
    \item \rot{Corte:} erro-padrão agrupado por unidade (Bertrand, Duflo e Mullainathan, 2004).
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{DiD, passo 3: os grupos já se afastavam antes?}
\begin{consoleR}
@@> estudo <- feols(renda ~ i(trimestre, tratado, ref = 3) |
@@+         id + trimestre, data = painel, cluster = ~id)
@@> round(coeftable(estudo)[c(1, 2, 3, 5), c(1, 4)], 2)
                     Estimate Pr(>|t|)
trimestre::1:tratado    23.15     0.11
trimestre::2:tratado    20.67     0.13
trimestre::4:tratado   317.24     0.00
trimestre::6:tratado   321.22     0.00
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} distância tratados $-$ controle por trimestre (base: o 3.º). \textbf{Antes} (1 e 2): 23 e 21, $p>0{,}05$. \textbf{Depois} (4 e 6): perto de 300.
    \item \rot{Corte:} pré-coeficientes pequenos, $p>0{,}05$; isso não prova o paralelismo (Roth, 2022).
  \end{itemize}
\end{leitura}
\end{frame}
