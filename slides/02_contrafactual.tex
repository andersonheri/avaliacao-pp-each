\secao{Bloco 1}{O problema do contrafactual}{14:15 -- 14:35 \,$\cdot$\, Script 01 \,$\cdot$\, Atividade 1}

\begin{frame}{O problema fundamental da inferência causal}
  Para cada pessoa, só observamos \destaque{um} dos dois mundos:
  \vspace{0.2cm}
  \begin{center}
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular}{lcccc}
    \toprule
    & \textbf{Participou?} & \textbf{Renda COM programa} & \textbf{Renda SEM programa} & \textbf{Efeito} \\ \midrule
    Ana    & sim & R\$ 2.400 & \destaque{?} & \destaque{?} \\
    Bruno  & não & \destaque{?} & R\$ 2.000 & \destaque{?} \\
    Carla  & sim & R\$ 2.700 & \destaque{?} & \destaque{?} \\
    Davi   & não & \destaque{?} & R\$ 1.800 & \destaque{?} \\ \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.3cm}
  \begin{itemize}
    \item O resultado que \emph{não} observamos é o \ind{contrafactual}: o que teria acontecido com a mesma pessoa no outro cenário;
    \item Toda técnica de avaliação de impacto é uma \destaque{estratégia para estimar o contrafactual}.
  \end{itemize}
  \fonte{Batista e Domingos (2017), p.~5; Gertler et al.~(2011).}
\end{frame}

\begin{frame}{O contrafactual: a comparação ideal}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \includegraphics[width=\textwidth]{01_contrafactual.png}
    \end{column}
    \begin{column}{0.45\textwidth}
      \small
      \begin{itemize}
        \item A comparação ideal é entre o grupo que recebeu a política e \emph{esse mesmo grupo} sem a política;
        \item Na vida real essa situação é \alert{inviável}: a linha tracejada nunca é observada;
        \item Só numa \ind{simulação} podemos desenhar as duas linhas --- e é por isso que vamos simular.
      \end{itemize}
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), Figura 2 (p.~6); simulação nossa (script 01).}
\end{frame}

\begin{frame}{Dois ``falsos contrafactuais'' intuitivos --- e equivocados}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{1. Participantes $\times$ não participantes}
        \small
        Compara quem recebeu a política com quem não recebeu, \textbf{depois} do programa.\\[0.2cm]
        \destaque{Problema}: \ind{viés de seleção}. Quem participa pode ser diferente de quem não participa (mais motivado, mais escolarizado\ldots).
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{2. Antes $\times$ depois}
        \small
        Compara os \textbf{mesmos} participantes antes e depois do programa.\\[0.2cm]
        \destaque{Problema}: \ind{tendência no tempo}. A renda pode ter subido por crescimento econômico, inflação, ciclo de vida\ldots
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.4cm}
  \begin{center}
    Nos dois casos, a comparação ``\emph{contamina}'' o efeito do programa com outros fatores.
  \end{center}
  \fonte{Batista e Domingos (2017), p.~5: ``falsos contrafactuais'' (\emph{counterfeit counterfactuals}).}
\end{frame}

\begin{frame}[fragile]{Hora do R: \texttt{01\_contrafactual\_vies\_selecao.R}}
  \begin{horaR}{scripts/01\_contrafactual\_vies\_selecao.R}
\begin{lstlisting}
source("scripts/00_setup.R")
dados <- gerar_populacao(n = 5000, efeito = 300)

# Falso contrafactual 1: participantes x nao participantes
mean(dados$renda_depois[dados$participa == 1]) -
  mean(dados$renda_depois[dados$participa == 0])      # 587

# Falso contrafactual 2: antes x depois (so participantes)
part <- dados[dados$participa == 1, ]
mean(part$renda_depois) - mean(part$renda_antes)      # 402
\end{lstlisting}
  \end{horaR}
  \vspace{0.2cm}
  {\small Rodem linha a linha (Ctrl+Enter) e comparem com a verdade: \destaque{300}.}
\end{frame}

\begin{frame}{Resultado: os dois falsos contrafactuais erram}
  \begin{center}
  \renewcommand{\arraystretch}{1.5}
  \begin{tabular}{lrr}
    \toprule
    \textbf{Método} & \textbf{Estimativa} & \textbf{Erro} \\ \midrule
    \rowcolor{verde!15}
    Verdade (efeito simulado) & R\$ 300 & --- \\
    Participantes $\times$ não participantes & R\$ 587 & \destaque{+287} (viés de seleção) \\
    Antes $\times$ depois & R\$ 402 & \destaque{+102} (tendência) \\ \bottomrule
  \end{tabular}
  \end{center}
  \vspace{0.4cm}
  \begin{block}{Voltando à enquete}
    \small
    Resposta (b): o efeito é \textbf{menor} que a diferença observada. Os R\$ 587 misturam o efeito do programa (R\$ 300) com o fato de que os participantes \emph{já ganhavam mais e eram mais motivados}.
  \end{block}
\end{frame}
