\begin{frame}{Experimentos reais}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Desparasitação no Quênia}
        \scriptsize
        75 escolas, 3 grupos de 25 (1998 a 2001), \textbf{implementação em fases}: todos recebem, em ordem sorteada.\\[2pt]
        Crianças tratadas: \ind{melhor saúde} e \ind{maior frequência escolar}.
      \end{block}
      {\tiny Miguel e Kremer (2004).}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Seguro Popular (México)}
        \scriptsize
        100 \emph{clusters} sorteados.\\[2pt]
        Gastos com saúde \ind{caíram 23\%} no grupo de tratamento, mas \alert{sem efeito substantivo} no uso de serviços médicos.
      \end{block}
      {\tiny King et al.\ (2009).}
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  \small Sortear pode parecer arbitrário ou antiético, mas com recursos escassos a \textbf{loteria} costuma ser a regra mais justa.
\end{frame}

\begin{frame}{Problemas no campo e poder estatístico}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \footnotesize
      \begin{itemize}
        \item \textbf{\emph{Compliance}}: sorteados não aderem. Na simulação (70\% de adesão): ITT = R\$ 193 (efeito da \emph{oferta}).
        \item \textbf{Atrito}: se saem \emph{os que não melhoraram}, o viés volta (R\$ 345).
        \item \textbf{\emph{Spillovers}}: o tratamento ``transborda'' para o controle.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\textwidth]{02_poder.png}\par
      \tiny Poder: precisamos de 34 pessoas por grupo para detectar R\$ 300, mas 291 para R\$ 100.
    \end{column}
  \end{columns}
\end{frame}

dados <- gerar_populacao(n = 5000, efeito = 300)

# 1. Sorteio: metade tratamento, metade controle
dados$tratamento <- sample(rep(0:1, length.out = nrow(dados)))
dados$renda_exp <- ifelse(dados$tratamento == 1,
                          dados$renda_depois_Y1, dados$renda_depois_Y0)

# 2. Impacto: diferenca de medias e regressao
t.test(renda_exp ~ tratamento, data = dados)
lm(renda_exp ~ tratamento, data = dados)       # ~ 295

# 3. Pessoas por grupo para detectar R$ 300 (poder 80%)?
power.t.test(delta = 300, sd = 430, power = 0.80)
