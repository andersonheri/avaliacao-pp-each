\begin{frame}{O ``padrão ouro'': a aleatorização}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \item Participantes são \ind{distribuídos aleatoriamente} em dois grupos:
        \begin{itemize}
          \item \textbf{Tratamento}: recebe a política;
          \item \textbf{Controle}: não recebe.
        \end{itemize}
        \item Todos têm a \emph{mesma probabilidade} de cair em cada grupo $\Rightarrow$ os grupos são equivalentes \destaque{em média}, nas características observadas \textbf{e não observadas};
        \item A única coisa que distingue os grupos é \alert{receber ou não a política}: a diferença nos resultados é o impacto.
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{atividadebox}{Atenção: o desenho vem antes}
        \small
        A avaliação experimental é definida no momento da \textbf{formulação} da política: é preciso medir o \emph{baseline} e sortear \emph{antes} de implementar.
      \end{atividadebox}
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), pp.~6--8.}
\end{frame}

\begin{frame}{Estimando o efeito: diferença de médias e regressão}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Diferença de médias}
        \small
        $\widehat{\text{Impacto}}=\bar Y_{\text{trat}}-\bar Y_{\text{ctrl}}$\\[0.15cm]
        Renda média: R\$ 2.328 $-$ R\$ 2.033 \\ $=$ \destaque{R\$ 295}
      \end{block}
      \begin{block}{Regressão}
        \small
        $Y_i=\beta_0+\beta_1\,\text{Trat}_i+\varepsilon_i$\\[0.15cm]
        $\hat\beta_1$ é o impacto.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{center}
        \small
        \renewcommand{\arraystretch}{1.4}
        \begin{tabular}{lrr}
          \toprule
          & \textbf{Efeito} & \textbf{Erro-padrão} \\ \midrule
          Sem covariáveis & 295,2 & 11,4 \\
          Com covariáveis & 302,1 & \destaque{3,4} \\ \bottomrule
        \end{tabular}
      \end{center}
      \vspace{0.2cm}
      \small Covariáveis não são necessárias para \emph{eliminar} o viés (o sorteio já fez isso), mas deixam a estimativa mais \ind{precisa}.\\[0.2cm]
      \textbf{Verdade: R\$ 300.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Experimentos na avaliação de políticas: dois exemplos}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Desparasitação no Quênia (PSDP)}
        \footnotesize
        \begin{itemize}
          \item 75 escolas divididas em 3 grupos de 25 (1998--2001);
          \item \textbf{Implementação em fases}: grupo 1 recebe no 1.º ano; grupo 2 no 2.º; grupo 3 no 3.º;
          \item Crianças tratadas tiveram \ind{melhor saúde} e \ind{maior frequência escolar}.
        \end{itemize}
      \end{block}
      \vspace{-0.1cm}
      {\tiny Miguel e Kremer (2004).}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Seguro Popular (México)}
        \footnotesize
        \begin{itemize}
          \item Acesso à saúde para famílias de baixa renda; 100 \emph{clusters} randomizados;
          \item Gastos com saúde \ind{caíram 23\%} no grupo de tratamento (55\% entre quem aderiu);
          \item Mas \alert{sem efeito substantivo} no uso de serviços médicos.
        \end{itemize}
      \end{block}
      \vspace{-0.1cm}
      {\tiny King et al.~(2009).}
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), p.~8.}
\end{frame}

\begin{frame}{Problemas no campo: \textit{compliance}, atrito e \textit{spillovers}}
  \begin{columns}[T]
    \begin{column}{0.33\textwidth}
      \begin{block}{\textit{Compliance}}
        \footnotesize
        Sorteados não aderem; ou o controle recebe o benefício por outro canal.\\[0.1cm]
        Na simulação (adesão de 70\%): \textbf{ITT = R\$ 193} (efeito da \emph{oferta}); efeito sobre quem aderiu $\approx$ R\$ 293.
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Atrito}
        \footnotesize
        Indivíduos ``saem'' do grupo de tratamento (mudança, não resposta).\\[0.1cm]
        Se saem de forma \emph{sistemática} (os que não melhoraram somem), o viés volta: estimativa de \textbf{R\$ 345} (verdade: 300).
      \end{block}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{\textit{Spillovers}}
        \footnotesize
        O tratamento ``transborda'' para o controle (ex.: vizinhos que compartilham informação).\\[0.1cm]
        O controle é contaminado e a diferença \emph{subestima} o efeito.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.3cm}
  \begin{center}\small Esses problemas dificultam a interpretação, mas \textbf{não a inviabilizam} (Khandker, Koolwal e Samad, 2010).\end{center}
  \fonte{Batista e Domingos (2017), p.~7; números da simulação (script 02).}
\end{frame}

\begin{frame}{Limites do experimento}
  \begin{itemize}
    \item \textbf{Viabilidade}: exige desenhar a avaliação \emph{antes} da implementação; se a política já foi distribuída de forma não aleatória, não há mais experimento;
    \item \textbf{Ética e aceitação}: sortear quem recebe pode parecer arbitrário ou antiético --- mas, com recursos escassos, a loteria costuma ser a regra mais justa;
    \item \textbf{Custo e escala}: amostras grandes, acompanhamento de campo;
    \item \textbf{Validade externa}: o efeito vale para a população e o contexto do experimento.
  \end{itemize}
  \vspace{0.3cm}
  \begin{block}{Quando não dá para sortear\ldots}
    \small As técnicas quase-experimentais tentam \emph{emular} um grupo de controle com dados observacionais. É o que veremos a seguir.
  \end{block}
  \fonte{Batista e Domingos (2017), pp.~7--8.}
\end{frame}

                          dados$renda_depois_Y1, dados$renda_depois_Y0)

# 2. Impacto: diferenca de medias, teste t e regressao
t.test(renda_exp ~ tratamento, data = dados)
lm(renda_exp ~ tratamento, data = dados)
