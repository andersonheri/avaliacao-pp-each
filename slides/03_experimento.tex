  \vspace{0.2cm}
  \begin{itemize}
    \item O sorteio torna os grupos equivalentes \destaque{em média}, nas características observadas e \textbf{não observadas};
    \item Só difere o receber ou não a política: a diferença nos resultados é o \ind{impacto}.
  \end{itemize}

\begin{frame}{Experimento: características e exemplos}
  \carac{grupo de \textbf{controle sorteado}}
        {sorteio bem executado; sem contaminação (\emph{spillovers}); atrito não seletivo}
        {efeito médio (ATE); ITT se a adesão é imperfeita}
        {desenho \textbf{antes} da política; amostra com poder estatístico}
        {recursos escassos, pilotos, expansão em fases}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{Quênia:} desparasitação em escolas, entrada sorteada por fases; mais frequência escolar (Miguel e Kremer, 2004).
      \item \textbf{México:} Progresa, localidades sorteadas para entrar primeiro (Gertler, 2004; Schultz, 2004); Seguro Popular, gastos com saúde caíram 23\% (King et al., 2009).
      \item \textbf{Hipotético (nosso Qualifica):} sorteio das vagas excedentes de um curso de qualificação municipal.
    \end{itemize}
\end{frame}

\subsection{Passo a passo}

\begin{frame}[fragile]{Experimento, passo 1: o sorteio equilibrou os grupos?}
\begin{consoleR}
@@> sapply(c("idade", "mulher", "escolaridade", "renda_antes"),
@@+        function(v) round(smd(dados[[v]], dados$tratamento), 3))
       idade       mulher escolaridade  renda_antes
       0.049        0.006        0.014       -0.018
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} um número por característica. É a \textbf{diferença entre as médias dos dois grupos, medida em desvios-padrão} (SMD). Zero = grupos iguais.
    \item \rot{Exemplo:} idade $=0{,}049$: os grupos diferem em 5\% de um desvio-padrão de idade (cerca de 7 meses). Praticamente nada.
    \item \rot{Regra prática:} $|\text{SMD}|<0{,}10$ é uma diferença pequena demais para distorcer a comparação (Austin, 2009).
    \item \rot{Conclusão:} todos ficaram abaixo de 0,05: o sorteio funcionou e podemos comparar os grupos.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Experimento, passo 2: quanto a renda aumentou?}
\begin{consoleRpeq}
@@> t.test(renda_exp ~ tratamento, data = dados)
    Welch Two Sample t-test
data:  renda_exp by tratamento
t = -25.859, df = 4994.8, p-value < 2.2e-16
alternative hypothesis: true difference in means between group 0 and group 1 is not equal to 0
95 percent confidence interval:
 -317.5637 -272.8064
sample estimates:
mean in group 0 mean in group 1
       2032.597        2327.782
\end{consoleRpeq}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Comece pelo fim:} renda média do controle (\texttt{group 0}) $=2.032{,}6$; do tratamento (\texttt{group 1}) $=2.327{,}8$. \textbf{Efeito} $=$ \destaque{R\$ 295}.
    \item \rot{Intervalo de confiança:} o efeito está, com 95\% de confiança, entre R\$ 273 e R\$ 318. O R mostra os números negativos porque faz ``controle $-$ tratamento''.
    \item \rot{p-valor} ($<2{,}2\times10^{-16}$): a chance de ver uma diferença desse tamanho \emph{se o programa não tivesse efeito}. Regra usual: $p<0{,}05$ = efeito significativo. Mas olhe o tamanho e o intervalo, não só o $p$ (Wasserstein e Lazar, 2016).
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Experimento, passo 3: dá para estimar com mais precisão?}
\begin{consoleR}
@@> m <- lm(renda_exp ~ tratamento + escolaridade + idade + mulher + renda_antes,
@@+         data = dados)
@@> round(summary(m)$coefficients, 3)
             Estimate Std. Error t value Pr(>|t|)
(Intercept)    83.452     10.431   8.000    0.000
tratamento    302.068      3.413  88.506    0.000
escolaridade   -0.853      0.815  -1.046    0.296
idade           0.116      0.143   0.816    0.414
mulher         -0.786      3.451  -0.228    0.820
renda_antes     1.011      0.006 158.389    0.000
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} só a linha \texttt{tratamento}. \texttt{Estimate} $=302{,}1$ é o efeito; \texttt{Std.\ Error} $=3{,}4$ é a margem de incerteza (era 11,4 no passo 2); \texttt{Pr(>|t|)} é o p-valor.
    \item \rot{Por que ficou mais preciso:} quem já ganhava mais antes continua ganhando mais; a regressão ``desconta'' isso e sobra menos ruído. O efeito quase não muda, porque o sorteio já tinha resolvido o viés.
    \item \rot{Cuidado:} as outras linhas \textbf{não} são efeitos de nada (Westreich e Greenland, 2013). Só use características medidas \emph{antes} do programa.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Experimento, passo 4: quantas pessoas preciso sortear?}
\begin{consoleR}
@@> power.t.test(delta = 100, sd = 430, sig.level = 0.05, power = 0.80)
     Two-sample t test power calculation
              n = 291.2151
          delta = 100
             sd = 430
      sig.level = 0.05
          power = 0.8
    alternative = two.sided
NOTE: n is number in *each* group
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Você informa:} \texttt{delta} $=100$ (o menor efeito que vale a pena detectar); \texttt{sd} $=430$ (quanto a renda varia entre as pessoas, de dados anteriores); \texttt{sig.level} $=0{,}05$ (aceito 5\% de risco de ver efeito onde não há); \texttt{power} $=0{,}80$ (quero 80\% de chance de ver o efeito, se ele existir).
    \item \rot{O R responde:} \texttt{n} $=291{,}2$: são \textbf{292 pessoas em cada grupo} (584 no total).
    \item \rot{Intuição:} R\$ 100 é pouco perto da variação da renda (430), então é preciso muita gente; para um efeito de R\$ 300 bastariam 34 por grupo. 80\% e 5\% são o padrão (Cohen, 1988).
  \end{itemize}
\end{leitura}
\end{frame}
