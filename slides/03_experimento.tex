\subsection{Passo a passo}

\begin{frame}[fragile]{Experimento, passo 1: o sorteio equilibrou os grupos?}
\begin{consoleR}
@@> sapply(c("idade", "mulher", "escolaridade", "renda_antes"),
@@+        function(v) round(smd(dados[[v]],
@@+                             dados$tratamento), 3))
       idade       mulher escolaridade  renda_antes
       0.049        0.006        0.014       -0.018
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} um número por característica: a diferença entre os grupos em desvios-padrão (SMD). Zero = iguais.
    \item \rot{Corte:} $|\text{SMD}|<0{,}10$ (Austin, 2009). Todos abaixo de 0,05: o sorteio funcionou.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Experimento, passo 2: quanto a renda aumentou?}
\begin{consoleR}
@@> m <- lm(renda_exp ~ tratamento, data = dados)
@@> round(c(coef(m)["tratamento"],
@@+         confint(m)["tratamento", ]), 1)
tratamento      2.5 %     97.5 %
     295.2      272.8      317.6
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} \texttt{tratamento} é o efeito: \destaque{R\$ 295} (verdade: 300). Os outros dois números são o IC de 95\%: de R\$ 273 a R\$ 318.
    \item \rot{Corte:} IC sem o zero $\Rightarrow$ efeito significativo ($p<0{,}05$). Olhe o tamanho e o intervalo, não só o $p$ (Wasserstein e Lazar, 2016).
  \end{itemize}
\end{leitura}
\end{frame}

\subsection{Experimentos naturais}

\begin{frame}{Experimentos naturais: quando a ``sorte'' faz o sorteio}
  \begin{itemize}
    \item Nem sempre dá para sortear. Às vezes uma \ind{regra}, um \ind{sorteio administrativo} ou um \ind{choque} divide as pessoas \emph{como se} fosse aleatório;
    \item O pesquisador \textbf{não desenha} a divisão: encontra-a e \textbf{argumenta} que ela é ``quase aleatória'';
    \item Vantagem: dados de políticas reais, em larga escala. Limite: a aleatoriedade \textbf{não é garantida}; precisa ser defendida e testada (Dunning, 2012).
  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{Pergunta-chave}
    \small Quem recebeu e quem não recebeu a política diferem \emph{só} pelo acaso da regra?
  \end{block}
\end{frame}

\end{frame}

\begin{frame}{Experimentos naturais em políticas públicas}
  \footnotesize
  \begin{itemize}
    \item \textbf{EUA:} loteria do alistamento militar (Vietnã) sorteou quem seria convocado e permitiu medir o efeito sobre a renda de veteranos (Angrist, 1990).
    \item \textbf{EUA:} Oregon sorteou vagas no Medicaid; efeitos sobre saúde e gastos (Finkelstein et al., 2012).
    \item \textbf{Brasil:} sorteio público de municípios para auditorias da CGU; efeito da divulgação de irregularidades sobre a reeleição de prefeitos (Ferraz e Finan, 2008).
    \item \textbf{Israel:} regra de Maimônides (turmas divididas acima de 40 alunos) e desempenho escolar (Angrist e Lavy, 1999).
  \end{itemize}
  \vspace{0.2cm}
  \begin{block}{Cuidados}
    \scriptsize Mostre que os grupos são parecidos \emph{antes}; descreva como o acaso operou; teste se houve manipulação da regra; diga \textbf{para quem} vale o efeito.
  \end{block}
\end{frame}
