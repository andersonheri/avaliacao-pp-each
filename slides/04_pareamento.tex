\begin{frame}{A maldição da dimensionalidade e o escore de propensão}
  \centering
  \includegraphics[height=3.9cm]{03_maldicao.png}\par
  \footnotesize Pareamento exato: cada nova variável derruba a chance de achar um ``gêmeo''. \textbf{Solução}: resumir tudo em um número, o \ind{propensity score} (probabilidade de participar), e parear por ele.
\end{frame}

\begin{frame}{Pareamento: características e exemplos}
  \carac{não participantes \textbf{parecidos} (mesmo escore de propensão)}
        {\textbf{sem confundidores não observados}; região comum (há semelhantes nos dois grupos)}
        {efeito sobre os tratados (ATT)}
        {muitas covariáveis \textbf{anteriores} ao programa (idealmente o resultado antes)}
        {política já implementada, sem sorteio, corte ou painel}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{Argentina:} programa de emprego Trabajar durante a crise de 1997 (Jalan e Ravallion, 2003).
      \item \textbf{Brasil:} Bolsa Família, efeitos sobre consumo, educação e saúde (MDS, 2007).
      \item \textbf{EUA:} programa de treinamento profissional reanalisado com PSM (LaLonde, 1986; Dehejia e Wahba, 1999).
      \item \textbf{Hipotético (nosso Qualifica):} programa já implementado, sem sorteio.
    \end{itemize}
\end{frame}

\subsection{Passo a passo}

\begin{frame}[fragile]{Pareamento, passo 1: estimar o escore de propensão}
\fazer{modelar a probabilidade de participar com covariáveis \textbf{pré-programa} (script 03).}
\begin{consoleR}
@@> m_ps <- glm(participa ~ escolaridade + idade + mulher + renda_antes,
@@+             data = dados, family = binomial)
@@> round(summary(m_ps)$coefficients, 4)
             Estimate Std. Error  z value Pr(>|z|)
(Intercept)   -4.6114     0.2185 -21.1034   0.0000
escolaridade  -0.0480     0.0155  -3.1063   0.0019
idade         -0.0235     0.0028  -8.4700   0.0000
mulher         0.1883     0.0660   2.8522   0.0043
renda_antes    0.0026     0.0001  19.5414   0.0000
\end{consoleR}
\begin{leitura}
  \begin{itemize}
    \item O sinal diz quem tende a participar: quem tinha \textbf{renda antes} maior ($+$) e é mais jovem ($-$ idade).
    \item O modelo serve para \textbf{balancear}, não para interpretar causas: não se lê esses coeficientes como efeitos (Rosenbaum e Rubin, 1983).
    \item Só entram variáveis medidas \textbf{antes} do programa.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Pareamento, passo 2: há região comum?}
\fazer{comparar a distribuição do escore entre não participantes (0) e participantes (1).}
\begin{consoleR}
@@> dados$ps <- predict(m_ps, type = "response")
@@> tapply(dados$ps, dados$participa, summary)
$`0`
   Min. 1st Qu.  Median    Mean 3rd Qu.    Max.
 0.0103  0.1592  0.2524  0.2797  0.3752  0.9055
$`1`
   Min. 1st Qu.  Median    Mean 3rd Qu.    Max.
0.04621 0.28872 0.41042 0.42225 0.55083 0.89851
\end{consoleR}
\begin{leitura}
  \begin{itemize}
    \item Participantes vão de 0,046 a 0,899; não participantes, de 0,010 a 0,906: as faixas se \textbf{sobrepõem} quase inteiras.
    \item Os participantes têm escore médio maior (0,42 $\times$ 0,28): é o viés de seleção, que o pareamento vai corrigir.
    \item \corte{com pouca sobreposição, descarte quem tem escore fora de [0,1; 0,9] (Crump et al., 2009).}
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Pareamento, passo 3: o pareamento equilibrou?}
\fazer{parear (vizinho mais próximo, \emph{caliper} de 0,2 dp) e comparar o balanço antes e depois.}
\begin{consoleR}
@@> sum(!is.na(par_controle)); length(tratados)
[1] 1630
[1] 1631
@@> round(balanco, 3)
             smd_antes smd_depois razao_var
escolaridade     0.502     -0.032     0.946
idade           -0.170      0.008     1.010
mulher           0.000      0.012     1.001
renda_antes      0.801     -0.004     0.967
\end{consoleR}
\begin{leitura}
  \begin{itemize}
    \item 1.630 dos 1.631 participantes acharam par (\emph{caliper} de 0,2 dp do logit do escore: Austin, 2011).
    \item \corte{$|\text{SMD}|<0{,}10$ (Austin, 2009) e razão de variâncias entre 0,5 e 2 (Rubin, 2001).} Antes: 0,50 e 0,80 (desbalanceado). Depois: todos ${<}\,0{,}04$ e razões ${\approx}\,1$.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Pareamento, passo 4: estimar o efeito (ATT)}
\fazer{média das diferenças de renda dentro dos pares; incerteza por reamostragem.}
\begin{consoleR}
@@> att_psm
[1] 297.3
@@> c(EP = sd(boot), quantile(boot, c(0.025, 0.975)))
   EP  2.5% 97.5%
  8.8 282.4 314.2
\end{consoleR}
\begin{leitura}
  \begin{itemize}
    \item Participantes ganham \destaque{R\$ 297} a mais por causa do programa (verdade: 300); IC 95\% de 282 a 314.
    \item Atenção: o \emph{bootstrap} simples não é válido para vizinho mais próximo (Abadie e Imbens, 2008); em trabalho real, use o erro-padrão de Abadie e Imbens.
    \item \textbf{Robustez:} sem a renda antes no escore, o ATT vira 476. O PSM só iguala o que está no modelo.
  \end{itemize}
\end{leitura}
\end{frame}
