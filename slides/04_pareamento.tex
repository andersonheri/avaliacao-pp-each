\begin{frame}{E quando o experimento não é possível?}
  \begin{itemize}
    \item A política \alert{já foi implementada}, de forma não aleatória: não há separação \emph{ex-ante} entre tratamento e controle;
    \item O que temos são \ind{dados observacionais}: quem participou, quem não participou, e características de ambos;
    \item As \ind{técnicas quase-experimentais} usam estatística para \destaque{emular} um grupo de controle.
  \end{itemize}
  \vspace{0.3cm}
  \begin{block}{Três técnicas que veremos}
    \small
    \begin{enumerate}
      \item \textbf{Pareamento} (\emph{matching}) e \emph{propensity score}: compara indivíduos \emph{parecidos};
      \item \textbf{Diferença-em-diferenças}: compara \emph{variações} ao longo do tempo;
      \item \textbf{Regressão descontínua}: explora uma \emph{regra de elegibilidade}.
    \end{enumerate}
  \end{block}
  \fonte{Batista e Domingos (2017), p.~8.}
\end{frame}

\begin{frame}{Pareamento exato e a maldição da dimensionalidade}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \includegraphics[width=\textwidth]{03_maldicao.png}
    \end{column}
    \begin{column}{0.5\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Pareamento exato}: o controle tem \emph{exatamente} os mesmos atributos;
        \item Parece ideal, mas com muitas variáveis (ou variáveis quase contínuas) \alert{quase ninguém tem gêmeo exato};
        \item Na simulação: de 100\% (com 2 variáveis) a \destaque{3\%} (renda medida em dezenas de reais).
      \end{itemize}
      \vspace{0.2cm}
      \begin{block}{Solução (Rosenbaum e Rubin, 1983)}
        \footnotesize Resumir tudo em \textbf{um número}: a \emph{probabilidade de participar}.
      \end{block}
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), pp.~8--9; simulação nossa (script 03).}
\end{frame}

\begin{frame}{Propensity Score Matching (PSM) em 3 passos}
  \centering
  \begin{tikzpicture}[node distance=0.7cm,scale=0.82,every node/.style={transform shape}]
    \node[cxi,text width=4.6cm] (p1) {\textbf{1. Estimar o propensity score}\\[2pt]\scriptsize Regressão logística: $P(\text{participar}\mid \text{características})$};
    \node[cxi,right=0.7cm of p1,text width=4.6cm] (p2) {\textbf{2. Checar a região comum}\\[2pt]\scriptsize Há participantes e não participantes com escores parecidos?};
    \node[cxi,right=0.7cm of p2,text width=4.6cm] (p3) {\textbf{3. Parear e comparar}\\[2pt]\scriptsize Vizinho mais próximo (com \emph{caliper}); efeito = diferença nos pares};
    \draw[seta] (p1) -- (p2); \draw[seta] (p2) -- (p3);
  \end{tikzpicture}
  \vspace{0.4cm}
  \begin{itemize}
    \item O pesquisador não encontra dois indivíduos idênticos, mas indivíduos \ind{semelhantes o suficiente} na propensão de participar;
    \item Depois do pareamento, é obrigatório \destaque{checar o balanceamento} das covariáveis.
  \end{itemize}
  \fonte{Batista e Domingos (2017), pp.~9--10.}
\end{frame}

\begin{frame}{Pressupostos e críticas ao PSM}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Pressupostos}
        \small
        \begin{enumerate}
          \item \textbf{Sem confundidores não observados}: fatores não medidos (ex.: autoestima, motivação) \emph{não} afetam a participação nem o resultado;
          \item \textbf{Região comum grande}: há indivíduos semelhantes participando e não participando.
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Críticas}
        \small
        \begin{itemize}
          \item King e Nielsen (2016): o PSM pode \alert{aumentar} o desbalanceamento, que cresce com o nº de variáveis;
          \item Alternativa: \textbf{Coarsened Exact Matching} (CEM; Iacus, King e Porro, 2011/2012), que \emph{garante} um desbalanceamento máximo escolhido \emph{ex-ante};
          \item Sempre: \ind{mostre o balanceamento}.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), pp.~9--11.}
\end{frame}

\begin{frame}{PSM na avaliação de políticas: dois exemplos}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{block}{Trabajar (Argentina, 1997)}
        \footnotesize
        Jalan e Ravallion (2003) analisam o programa argentino durante a crise econômica de 1997 e examinam os \textbf{ganhos de renda} dos beneficiários. O PSM usa variáveis socioeconômicas, do domicílio, demográficas e de histórico educacional. Com os escores estimados, separam tratamento e controle e \alert{não encontram efeitos positivos} do programa.
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Bolsa Família (Brasil)}
        \footnotesize
        MDS (2007): impacto sobre gasto de consumo, educação, trabalho infantil, antropometria, saúde e poder de barganha das mulheres. Dois grupos de comparação: elegíveis que recebem \emph{outros} programas e não elegíveis; os participantes tiveram \textbf{melhor desempenho}.
      \end{block}
    \end{column}
  \end{columns}
  \fonte{Batista e Domingos (2017), p.~10.}
\end{frame}

\begin{frame}[fragile]{Hora do R: \texttt{03\_pareamento\_psm.R}}

\begin{lstlisting}
# 1. Propensity score (regressao logistica)
m <- glm(participa ~ escolaridade + idade + mulher + renda_antes,
         data = dados, family = binomial)
dados$lps <- qlogis(predict(m, type = "response"))

# 2. Para cada tratado, o controle mais proximo (com caliper)
caliper <- 0.2 * sd(dados$lps)
for (k in seq_along(tratados)) {
  d <- abs(dados$lps[controles] - dados$lps[tratados[k]])
  if (min(d) <= caliper) par_controle[k] <- controles[which.min(d)]
}

# 3. Efeito sobre os tratados (ATT) e balanceamento (SMD)
mean(renda_tratados - renda_pares)                 # 297
\end{lstlisting}
  \end{horaR}
