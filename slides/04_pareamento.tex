\begin{frame}{A maldição da dimensionalidade e o escore de propensão}
  \centering
  \includegraphics[height=3.9cm]{03_maldicao.png}\par
  \footnotesize Pareamento exato: cada nova variável derruba a chance de achar um ``gêmeo''. \textbf{Solução}: resumir tudo em um número, o \ind{propensity score} (probabilidade de participar), e parear por ele.
\end{frame}

\begin{frame}{Pareamento: características e exemplos}
  \carac{não participantes \textbf{parecidos} (mesmo escore de propensão)}
        {\textbf{sem confundidores não observados}; região comum (há semelhantes nos dois grupos)}
        {efeito sobre os tratados (ATT)}
        {muitas covariáveis \textbf{anteriores} ao programa (idealmente o resultado antes)}
        {política já implementada, sem sorteio, corte ou painel}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{Argentina:} programa de emprego Trabajar durante a crise de 1997 (Jalan e Ravallion, 2003).
      \item \textbf{Brasil:} Bolsa Família, efeitos sobre consumo, educação e saúde (MDS, 2007).
      \item \textbf{EUA:} programa de treinamento profissional reanalisado com PSM (LaLonde, 1986; Dehejia e Wahba, 1999).
    \end{itemize}
\end{frame}

\subsection{Passo a passo}

\begin{frame}[fragile]{Pareamento, passo 1: existe com quem comparar?}
\begin{consoleR}
@@> modelo_ps <- glm(participa ~ escolaridade + idade + mulher +
@@+                  renda_antes, data = dados, family = binomial)
@@> dados$ps <- predict(modelo_ps, type = "response")
@@> round(sapply(split(dados$ps, dados$participa), range), 2)
        0    1
[1,] 0.01 0.05
[2,] 0.91 0.90
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} o escore é a probabilidade de participar. Colunas: \texttt{0} não participantes, \texttt{1} participantes; linhas: mínimo e máximo. As faixas \textbf{se sobrepõem} (0,05 a 0,90 contra 0,01 a 0,91): há com quem comparar.
    \item \rot{Corte:} excluir escores abaixo de 0,1 ou acima de 0,9 se faltar sobreposição (Crump et al., 2009).
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Pareamento, passo 2: os pares ficaram parecidos?}
\begin{consoleR}
@@> balanco_psm[, 2:3]
              antes depois
escolaridade  0.502 -0.032
idade        -0.170  0.008
mulher        0.000  0.012
renda_antes   0.801 -0.004
motivacao     0.763  0.303
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} \texttt{antes} e \texttt{depois} são o SMD antes e depois de parear. As quatro variáveis do modelo caíram a perto de zero.
    \item \rot{Corte:} $|\text{SMD}|<0{,}10$ (Austin, 2009). \texttt{motivacao} (não observada, fora do modelo) ficou em 0,30: o PSM só iguala o que está no modelo.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Pareamento, passo 3: qual o efeito e quão certo ele é?}
\begin{consoleR}
@@> round(att_a, 1)
[1] 297.3
@@> round(quantile(boot_a, c(0.025, 0.975)), 0)
 2.5% 97.5%
  282   314
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} \texttt{att\_a} é o efeito para quem participou: \destaque{R\$ 297} (verdade: 300). A 2.ª linha é o IC de 95\% por \emph{bootstrap}: R\$ 282 a R\$ 314.
    \item \rot{Corte:} IC sem o zero. Nota: para o vizinho mais próximo, o erro-padrão recomendado é o de Abadie e Imbens (2008); o \emph{bootstrap} é mais intuitivo.
  \end{itemize}
\end{leitura}
\end{frame}
