# 1. Propensity score (regressao logistica)
m <- glm(participa ~ escolaridade + idade + mulher + renda_antes,
         data = dados, family = binomial)
dados$lps <- qlogis(predict(m, type = "response"))

# 2. Para cada tratado, o controle mais proximo (com caliper)
caliper <- 0.2 * sd(dados$lps)
for (k in seq_along(tratados)) {
  d <- abs(dados$lps[controles] - dados$lps[tratados[k]])
  if (min(d) <= caliper) par_controle[k] <- controles[which.min(d)]
}

# 3. Efeito sobre os tratados (ATT)
