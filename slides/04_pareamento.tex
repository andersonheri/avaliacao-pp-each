\begin{frame}{A maldição da dimensionalidade e o escore de propensão}
  \centering
  \includegraphics[height=3.9cm]{03_maldicao.png}\par
  \footnotesize Pareamento exato: cada nova variável derruba a chance de achar um ``gêmeo''. \textbf{Solução}: resumir tudo em um número, o \ind{propensity score} (probabilidade de participar), e parear por ele.
\end{frame}

\begin{frame}{Pareamento: características e exemplos}
  \carac{não participantes \textbf{parecidos} (mesmo escore de propensão)}
        {\textbf{sem confundidores não observados}; região comum (há semelhantes nos dois grupos)}
        {efeito sobre os tratados (ATT)}
        {muitas covariáveis \textbf{anteriores} ao programa (idealmente o resultado antes)}
        {política já implementada, sem sorteio, corte ou painel}
  \vspace{0.05cm}
  \textbf{\footnotesize\color{azul}Exemplos em políticas públicas}\par
    \scriptsize
    \begin{itemize}
      \item \textbf{Argentina:} programa de emprego Trabajar durante a crise de 1997 (Jalan e Ravallion, 2003).
      \item \textbf{Brasil:} Bolsa Família, efeitos sobre consumo, educação e saúde (MDS, 2007).
      \item \textbf{EUA:} programa de treinamento profissional reanalisado com PSM (LaLonde, 1986; Dehejia e Wahba, 1999).
      \item \textbf{Hipotético (nosso Qualifica):} programa já implementado, sem sorteio.
    \end{itemize}
\end{frame}

\subsection{Passo a passo}

\begin{frame}[fragile]{Pareamento, passo 1: quem tende a se inscrever?}
\begin{consoleR}
@@> m_ps <- glm(participa ~ escolaridade + idade + mulher + renda_antes,
@@+             data = dados, family = binomial)
@@> round(summary(m_ps)$coefficients, 4)
             Estimate Std. Error  z value Pr(>|z|)
(Intercept)   -4.6114     0.2185 -21.1034   0.0000
escolaridade  -0.0480     0.0155  -3.1063   0.0019
idade         -0.0235     0.0028  -8.4700   0.0000
mulher         0.1883     0.0660   2.8522   0.0043
renda_antes    0.0026     0.0001  19.5414   0.0000
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} o \textbf{sinal} da coluna \texttt{Estimate}. Positivo: aumenta a chance de participar; negativo: diminui.
    \item \rot{O que diz:} \texttt{renda\_antes} $+$: quem já ganhava mais se inscreveu mais; \texttt{idade} $-$: os mais jovens se inscreveram mais; \texttt{mulher} $+$: mulheres um pouco mais. Ninguém ``escolheu'' essas pessoas: \textbf{elas se escolheram} (autosseleção).
    \item \rot{Por que importa:} se quem já ganhava mais participou mais, a comparação simples exagera o efeito. O modelo dá a cada pessoa uma \textbf{probabilidade de participar} (o escore), para comparar participantes e não participantes com a \emph{mesma} probabilidade.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Pareamento, passo 2: existe com quem comparar?}
\begin{consoleR}
@@> dados$ps <- predict(m_ps, type = "response")
@@> tapply(dados$ps, dados$participa, summary)
$`0`
   Min. 1st Qu.  Median    Mean 3rd Qu.    Max.
 0.0103  0.1592  0.2524  0.2797  0.3752  0.9055
$`1`
   Min. 1st Qu.  Median    Mean 3rd Qu.    Max.
0.04621 0.28872 0.41042 0.42225 0.55083 0.89851
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Onde olhar:} \texttt{\$`0`} são os não participantes; \texttt{\$`1`}, os participantes. Compare \texttt{Min.} e \texttt{Max.}: participantes vão de 0,05 a 0,90 e não participantes de 0,01 a 0,91. As faixas \textbf{se sobrepõem}: há com quem comparar.
    \item \rot{Por que pode faltar comparação:} imagine alguém com escore 0,98: quase todo mundo com aquele perfil participou, então não sobra ninguém parecido entre os não participantes. Comparar seria ``chutar''.
    \item \rot{Regra prática:} nesse caso, excluir quem tem escore abaixo de 0,1 ou acima de 0,9 e dizer para quem o efeito vale (Crump et al., 2009). Aqui a sobreposição é boa: não precisamos cortar.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Pareamento, passo 3: os pares ficaram parecidos?}
\begin{consoleR}
@@> sum(!is.na(par_controle)); length(tratados)
[1] 1630
[1] 1631
@@> round(balanco, 3)
             smd_antes smd_depois razao_var
escolaridade     0.502     -0.032     0.946
idade           -0.170      0.008     1.010
mulher           0.000      0.012     1.001
renda_antes      0.801     -0.004     0.967
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{Primeiras linhas:} 1.630 dos 1.631 participantes acharam um par (só 1 ficou sem).
    \item \rot{Tabela:} \texttt{smd\_antes} é a diferença entre os grupos \emph{antes} de parear (renda antes: 0,80, muito diferente); \texttt{smd\_depois}, \emph{depois} ($-0{,}004$, praticamente igual). \texttt{razao\_var} compara o quanto cada grupo varia (1 = igual).
    \item \rot{Regras:} $|\text{SMD}|<0{,}10$ (Austin, 2009) e razão de variâncias entre 0,5 e 2 (Rubin, 2001): médias \emph{e} dispersão parecidas. Tudo dentro: pareamento bem-sucedido.
  \end{itemize}
\end{leitura}
\end{frame}

\begin{frame}[fragile]{Pareamento, passo 4: qual o efeito e quão certo ele é?}
\begin{consoleR}
@@> att_psm
[1] 297.3
@@> c(EP = sd(boot), quantile(boot, c(0.025, 0.975)))
   EP  2.5% 97.5%
  8.8 282.4 314.2
\end{consoleR}
\begin{leitura}
  \begin{itemize}\setlength{\itemsep}{1pt}
    \item \rot{\texttt{att\_psm} = 297,3:} a média da diferença de renda entre cada participante e o seu par: o efeito do programa para quem participou (verdade: 300).
    \item \rot{De onde vem \texttt{boot}:} refizemos todo o pareamento 100 vezes, cada vez numa amostra ``sorteada'' a partir da nossa (\emph{bootstrap}). Isso simula outras pesquisas possíveis.
    \item \rot{O que o comando mostra:} \texttt{sd(boot)} $=8{,}8$: quanto as 100 estimativas variam, ou seja, o \textbf{erro-padrão}. \texttt{quantile} 2,5\% e 97,5\%: entre esses valores ficam 95\% das estimativas, ou seja, o \textbf{IC 95\%}: R\$ 282 a R\$ 314.
  \end{itemize}
\end{leitura}
\end{frame}
